159 BinaryExpr(BinaryOp op, std::unique_ptr<Expr> lhs,

169
170 // calcualte the value, when need
171 ConstValue to_const(ConstValue lhs, ConstValue rhs, BinaryOp type_) {
172 if (type_ == BinaryOp::And) return ConstValue((int)GetValue(lhs) & (int)GetValue(rhs));
173 if (type_ == BinaryOp::Or ) return ConstValue((int)GetValue(lhs) | (int)GetValue(rhs));
174
175 if (lhs.type == Float || rhs.type == Float) {
176 if (type_ == BinaryOp::Add) return ConstValue(GetValue(lhs) + GetValue(rhs));
177 if (type_ == BinaryOp::Sub) return ConstValue(GetValue(lhs) - GetValue(rhs));
178 if (type_ == BinaryOp::Mul) return ConstValue(GetValue(lhs) * GetValue(rhs));
179 if (type_ == BinaryOp::Div) return ConstValue(GetValue(lhs) / GetValue(rhs));
180 }
181 else {
182 if (type_ == BinaryOp::Add) return ConstValue((int)GetValue(lhs) + (int)GetValue(rhs));
183 if (type_ == BinaryOp::Sub) return ConstValue((int)GetValue(lhs) - (int)GetValue(rhs));
184 if (type_ == BinaryOp::Mul) return ConstValue((int)GetValue(lhs) * (int)GetValue(rhs));
185 if (type_ == BinaryOp::Div) return ConstValue((int)GetValue(lhs) / (int)GetValue(rhs));
186 }
187 if (type_ == BinaryOp::Mod) return ConstValue((int)GetValue(lhs) % (int)GetValue(rhs));
188
189 if (type_ == BinaryOp::Eq) return lhs == rhs;
190 if (type_ == BinaryOp::Neq) return lhs != rhs;
191
192 if (type_ == BinaryOp::Lt) return ConstValue(GetValue(lhs) < GetValue(rhs));
193 if (type_ == BinaryOp::Leq) return ConstValue(GetValue(lhs) <= GetValue(rhs));
194 if (type_ == BinaryOp::Gt) return ConstValue(GetValue(lhs) > GetValue(rhs));
195 if (type_ == BinaryOp::Geq) return ConstValue(GetValue(lhs) >= GetValue(rhs));
196 if (type_ == BinaryOp::Shr) return ConstValue((int)GetValue(lhs) >> (int)GetValue(rhs));
197 if (type_ == BinaryOp::Shl) return ConstValue((int)GetValue(lhs) << (int)GetValue(rhs));
198 assert(false);
199}
